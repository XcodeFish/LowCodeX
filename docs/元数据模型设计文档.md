# 元数据模型设计文档

## 1. 概述

本文档详细描述低代码平台中元数据模型的设计，包括元表(Meta Table)和元字段(Meta Field)的数据结构。这些结构是动态数据模型的基础，支持低代码平台的核心功能，如动态建表、字段配置、关系管理等。

## 2. 元数据模型概念

元数据模型是描述数据模型的数据结构，主要包括两个核心部分：

- **元表(Meta Table)**：描述业务实体的数据表结构
- **元字段(Meta Field)**：描述表中每个字段的结构和属性

## 3. 数据结构设计

### 3.1 元表(Meta Table)结构

```typescript
interface MetaTable {
  id: string;                 // 唯一标识符
  name: string;               // 技术名称（英文，用于数据库表名）
  displayName: string;        // 显示名称（中文，用于UI展示）
  description?: string;       // 表描述
  isSystem: boolean;          // 是否系统表
  isSoftDelete: boolean;      // 是否支持软删除
  isVersioned: boolean;       // 是否支持版本控制
  status: TableStatus;        // 表状态（设计中、已发布、已废弃等）
  tenant: string;             // 所属租户
  application?: string;       // 所属应用
  createdBy: string;          // 创建人
  createdAt: Date;            // 创建时间
  updatedBy?: string;         // 更新人
  updatedAt?: Date;           // 更新时间
  indexes?: MetaIndex[];      // 索引配置
  constraints?: MetaConstraint[]; // 约束配置
  auditFields: boolean;       // 是否包含审计字段（创建时间等）
  apiEnabled: boolean;        // 是否启用API
  customOptions?: Record<string, any>; // 自定义选项
}

enum TableStatus {
  DRAFT = 'draft',            // 草稿
  PUBLISHED = 'published',    // 已发布
  DEPRECATED = 'deprecated',  // 已废弃
  DELETED = 'deleted'         // 已删除
}

interface MetaIndex {
  id: string;                 // 索引ID
  name: string;               // 索引名称
  fields: string[];           // 索引字段 (字段名数组)
  type: IndexType;            // 索引类型
  isUnique: boolean;          // 是否唯一索引
}

enum IndexType {
  BTREE = 'btree',            // B树索引
  HASH = 'hash',              // 哈希索引
  FULLTEXT = 'fulltext'       // 全文索引
}

interface MetaConstraint {
  id: string;                 // 约束ID
  name: string;               // 约束名称
  type: ConstraintType;       // 约束类型
  fields: string[];           // 相关字段
  expression?: string;        // 约束表达式
  message?: string;           // 违反约束时的错误消息
}

enum ConstraintType {
  PRIMARY_KEY = 'primary_key',// 主键
  FOREIGN_KEY = 'foreign_key',// 外键
  UNIQUE = 'unique',          // 唯一性
  CHECK = 'check',            // 检查约束
  NOT_NULL = 'not_null'       // 非空约束
}
```

### 3.2 元字段(Meta Field)结构

```typescript
interface MetaField {
  id: string;                 // 唯一标识符
  tableId: string;            // 所属表ID
  name: string;               // 技术名称（英文）
  displayName: string;        // 显示名称（中文）
  description?: string;       // 字段描述
  type: FieldType;            // 字段类型
  isPrimaryKey: boolean;      // 是否主键
  isRequired: boolean;        // 是否必填
  isUnique: boolean;          // 是否唯一
  isSystem: boolean;          // 是否系统字段
  isHidden: boolean;          // 是否在UI中隐藏
  ordinal: number;            // 字段顺序
  defaultValue?: any;         // 默认值
  validationRules?: ValidationRule[]; // 验证规则
  isSearchable: boolean;      // 是否可搜索
  isSortable: boolean;        // 是否可排序
  isFilterable: boolean;      // 是否可筛选
  isAggregatable: boolean;    // 是否可聚合
  advancedSettings?: FieldAdvancedSettings; // 高级设置
}

enum FieldType {
  STRING = 'string',          // 字符串
  TEXT = 'text',              // 文本
  RICH_TEXT = 'richText',     // 富文本
  INTEGER = 'integer',        // 整数
  FLOAT = 'float',            // 浮点数
  DECIMAL = 'decimal',        // 精确小数
  BOOLEAN = 'boolean',        // 布尔值
  DATE = 'date',              // 日期
  DATETIME = 'datetime',      // 日期时间
  TIME = 'time',              // 时间
  ENUM = 'enum',              // 枚举
  JSON = 'json',              // JSON
  ARRAY = 'array',            // 数组
  REFERENCE = 'reference',    // 引用(关系)
  FILE = 'file',              // 文件
  IMAGE = 'image',            // 图片
  EMAIL = 'email',            // 电子邮件
  URL = 'url',                // 网址
  PHONE = 'phone',            // 电话号码
  COLOR = 'color',            // 颜色
  GEO = 'geo'                 // 地理位置
}

interface ValidationRule {
  type: ValidationRuleType;   // 规则类型
  message: string;            // 错误消息
  expression?: string;        // 表达式(用于自定义规则)
  parameters?: Record<string, any>; // 规则参数
}

enum ValidationRuleType {
  REQUIRED = 'required',      // 必填
  LENGTH = 'length',          // 长度
  RANGE = 'range',            // 范围
  REGEX = 'regex',            // 正则表达式
  CUSTOM = 'custom'           // 自定义规则
}

interface FieldAdvancedSettings {
  // 字符串类型设置
  minLength?: number;         // 最小长度
  maxLength?: number;         // 最大长度

  // 数字类型设置
  min?: number;               // 最小值
  max?: number;               // 最大值
  precision?: number;         // 精度(总位数)
  scale?: number;             // 小数位数

  // 枚举类型设置
  enumOptions?: EnumOption[]; // 枚举选项

  // 引用类型设置
  referenceTableId?: string;  // 引用的表ID
  referenceFieldId?: string;  // 引用的字段ID
  relationType?: RelationType;// 关系类型

  // UI渲染设置
  renderType?: string;        // 渲染类型
  renderOptions?: Record<string, any>; // 渲染选项

  // 业务规则设置
  businessRules?: BusinessRule[]; // 业务规则
}

interface EnumOption {
  value: string;              // 枚举值
  label: string;              // 显示标签
  color?: string;             // 颜色(用于UI展示)
  order?: number;             // 排序
  disabled?: boolean;         // 是否禁用
}

enum RelationType {
  ONE_TO_ONE = 'oneToOne',    // 一对一
  ONE_TO_MANY = 'oneToMany',  // 一对多
  MANY_TO_ONE = 'manyToOne',  // 多对一
  MANY_TO_MANY = 'manyToMany' // 多对多
}

interface BusinessRule {
  id: string;                 // 规则ID
  name: string;               // 规则名称
  type: BusinessRuleType;     // 规则类型
  condition?: string;         // 触发条件
  action: string;             // 执行操作
  priority: number;           // 优先级
  isActive: boolean;          // 是否激活
}

enum BusinessRuleType {
  VALIDATION = 'validation',  // 验证
  CALCULATION = 'calculation',// 计算
  AUTOMATION = 'automation',  // 自动化
  NOTIFICATION = 'notification' // 通知
}
```

### 3.3 元关系(Meta Relation)结构

```typescript
interface MetaRelation {
  id: string;                 // 关系ID
  name: string;               // 关系名称
  description?: string;       // 关系描述
  sourceTableId: string;      // 源表ID
  targetTableId: string;      // 目标表ID
  sourceFieldId: string;      // 源字段ID
  targetFieldId: string;      // 目标字段ID
  type: RelationType;         // 关系类型
  cascadeDelete: boolean;     // 是否级联删除
  cascadeUpdate: boolean;     // 是否级联更新
  isRequired: boolean;        // 是否必需关系
  junctionTableId?: string;   // 中间表ID(用于多对多)
  customOptions?: Record<string, any>; // 自定义选项
}
```

### 3.4 元版本(Meta Version)结构

```typescript
interface MetaVersion {
  id: string;                 // 版本ID
  tableId: string;            // 表ID
  version: number;            // 版本号
  name: string;               // 版本名称
  description?: string;       // 版本描述
  snapshot: MetaTable;        // 表快照
  fields: MetaField[];        // 字段快照
  relations: MetaRelation[];  // 关系快照
  isPublished: boolean;       // 是否已发布
  createdBy: string;          // 创建人
  createdAt: Date;            // 创建时间
  comment?: string;           // 版本说明
}
```

## 4. 数据关系

1. **元表与元字段**：一对多关系，一个元表包含多个元字段
2. **元表与元关系**：一对多关系，一个元表可以有多个与其相关的关系
3. **元表与元版本**：一对多关系，一个元表可以有多个版本记录

## 5. 功能场景

### 5.1 模型设计

- 创建、编辑、删除数据模型
- 定义字段类型、验证规则、关系
- 可视化ER图设计

### 5.2 版本控制

- 模型版本管理
- 版本比较和回滚
- 版本发布管理

### 5.3 数据服务生成

- 基于元数据自动生成API
- 自动创建数据库表结构
- 权限控制集成

### 5.4 UI组件绑定

- 表单控件自动生成
- 列表/详情页面生成
- 高级查询界面生成

## 6. 技术实现注意事项

1. **类型安全**：使用TypeScript确保类型定义严谨
2. **性能优化**：考虑元数据的缓存策略
3. **可扩展性**：设计时预留扩展点，支持自定义字段类型
4. **数据验证**：实现完整的输入验证保证元数据一致性
5. **并发控制**：实现乐观锁或版本控制机制防止并发修改冲突

## 7. 存储实现建议

### 7.1 数据库表设计

#### meta_tables表

```sql
CREATE TABLE meta_tables (
  id VARCHAR(36) PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  display_name VARCHAR(100) NOT NULL,
  description TEXT,
  is_system BOOLEAN NOT NULL DEFAULT FALSE,
  is_soft_delete BOOLEAN NOT NULL DEFAULT TRUE,
  is_versioned BOOLEAN NOT NULL DEFAULT FALSE,
  status VARCHAR(20) NOT NULL DEFAULT 'draft',
  tenant VARCHAR(36) NOT NULL,
  application VARCHAR(36),
  created_by VARCHAR(36) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_by VARCHAR(36),
  updated_at TIMESTAMP,
  audit_fields BOOLEAN NOT NULL DEFAULT TRUE,
  api_enabled BOOLEAN NOT NULL DEFAULT TRUE,
  custom_options JSON,
  UNIQUE (name, tenant, application)
);
```

#### meta_fields表

```sql
CREATE TABLE meta_fields (
  id VARCHAR(36) PRIMARY KEY,
  table_id VARCHAR(36) NOT NULL REFERENCES meta_tables(id),
  name VARCHAR(100) NOT NULL,
  display_name VARCHAR(100) NOT NULL,
  description TEXT,
  type VARCHAR(20) NOT NULL,
  is_primary_key BOOLEAN NOT NULL DEFAULT FALSE,
  is_required BOOLEAN NOT NULL DEFAULT FALSE,
  is_unique BOOLEAN NOT NULL DEFAULT FALSE,
  is_system BOOLEAN NOT NULL DEFAULT FALSE,
  is_hidden BOOLEAN NOT NULL DEFAULT FALSE,
  ordinal INT NOT NULL,
  default_value TEXT,
  validation_rules JSON,
  is_searchable BOOLEAN NOT NULL DEFAULT FALSE,
  is_sortable BOOLEAN NOT NULL DEFAULT FALSE,
  is_filterable BOOLEAN NOT NULL DEFAULT FALSE,
  is_aggregatable BOOLEAN NOT NULL DEFAULT FALSE,
  advanced_settings JSON,
  UNIQUE (table_id, name)
);
```

#### meta_relations表

```sql
CREATE TABLE meta_relations (
  id VARCHAR(36) PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  description TEXT,
  source_table_id VARCHAR(36) NOT NULL REFERENCES meta_tables(id),
  target_table_id VARCHAR(36) NOT NULL REFERENCES meta_tables(id),
  source_field_id VARCHAR(36) NOT NULL REFERENCES meta_fields(id),
  target_field_id VARCHAR(36) NOT NULL REFERENCES meta_fields(id),
  type VARCHAR(20) NOT NULL,
  cascade_delete BOOLEAN NOT NULL DEFAULT FALSE,
  cascade_update BOOLEAN NOT NULL DEFAULT FALSE,
  is_required BOOLEAN NOT NULL DEFAULT FALSE,
  junction_table_id VARCHAR(36) REFERENCES meta_tables(id),
  custom_options JSON
);
```

#### meta_versions表

```sql
CREATE TABLE meta_versions (
  id VARCHAR(36) PRIMARY KEY,
  table_id VARCHAR(36) NOT NULL REFERENCES meta_tables(id),
  version INT NOT NULL,
  name VARCHAR(100) NOT NULL,
  description TEXT,
  snapshot JSON NOT NULL,
  is_published BOOLEAN NOT NULL DEFAULT FALSE,
  created_by VARCHAR(36) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  comment TEXT,
  UNIQUE (table_id, version)
);
```
